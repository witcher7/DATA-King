from pyspark.sql.functions import max

emp_data = [
    ("Manish","1",75000),
    ("Neha","2",80000),
    ("Suresh","1",90000),
    ("Priya","3",85000),
    ("Ravi","3",95000),
    ("Sunita","1",88000),
    ("Rahul","2",72000)
]

emp_schema = ["name","dept_id","salary"]

dept_data = [
    ("1","IT"),
    ("2","HR"),
    ("3","Finance")
]

dept_schema = ["dept_id","dept_name"]

emp_df = spark.createDataFrame(emp_data, emp_schema)
dept_df = spark.createDataFrame(dept_data, dept_schema)

df = emp_df.join(dept_df, "dept_id", "left")

max_salary_df = df.groupBy("dept_name").agg(
    max("salary").alias("max_salary")
)

result = (
    df.join(max_salary_df, "dept_name", "inner")
      .filter(df.salary == max_salary_df.max_salary)
      .select("name", df.dept_name.alias("dept_name"), "salary")
)

result.show()

data = [
    ("Alice", "Math", 85),
    ("Alice", "Science", 90),
    ("Alice", "English", 88),
    ("Bob", "Math", 78),
    ("Bob", "Science", 82),
    ("Bob", "English", 80)
]

columns = ["name", "subject", "score"]

df = spark.createDataFrame(data, columns)

# Group by name and collect subjects and scores as arrays
result = df.groupBy("name").agg(
    collect_list("subject").alias("subjects"),
    collect_list("score").alias("scores")
)

result.show(truncate=False)

Find out the cummulative sales or running total of sales

> data = [("2023-01-01", 100), ("2023-01-02", 200), ("2023-01-03", 150)]
> df = spark.createDataFrame(data, ["date", "sales"])
> from pyspark.sql.window import Window
> from pyspark.sql.functions import sum
> window = Window.orderBy("date")
> df.withColumn("cumulative_sales", sum("sales").over(window)).show()

> from pyspark.sql.functions import lag
> window = Window.orderBy("date")
> df.withColumn("previous_sales", lag("sales", 1).over(window)).show()
