         -> spark.read
         -> spark.read.format("csv")
         -> spark.read.option("header", "true")
         -> spark.read.option("inferSchema", "true")
         -> spark.read.option("mode", "DROPMALFORMED"/"PERMISSIVE"/"FAILFAST")
         -> spark.read.schema("schema")
         -> spark.read.load("path")

FailFast => throws exception if there is a bad record
DropMalformed => drops the bad records
Permissive => sets all the fields to null if there is a bad record
-> Permissive is default mode

#### How to create dataframe
Dataframe is two dimensional structure with rows and columns and stores potentially large amount of data.
it is mutable like RDD.

df = spark.read.csv("path")
df = spark.read.json("path")
df = spark.read.parquet("path")
df = spark.read.jdbc("path")
df = spark.read.text("path")

> df.show() --> to show the dataframe 
> display(df) --> to show the dataframe in a table format
> df.show(10) --> to show the first 10 rows
> df.show(10, truncate=False) --> to show the first 10 rows without truncating
> df.show(10, truncate=50) --> to show the first 10 rows with 50 characters truncated

Dataframe with header
df1 = spark.read.format("csv").option("header", "true").load("path")
df1.show()
or 
df1 = spark.read.csv("path", header=True)
# the difference is that format() is more explicit and load() is more flexible and can be used for any format

> To see the schema of the dataframe
df1.printSchema()

> Dataframe with header and proper schema
df2 = spark.read.format("csv").option("header", "true").option("inferSchema", "true").load("path")
df2.printSchema()



## How to read JSON file 
df3 = spark.read.format("json").load("path")

## to read multiple lines JSON 
df4 = spark.read.format("json").option("multiLine", "true").load("path")

## How to select specific columns
df5 = df.select("column1", "column2")
df5.show()

## also 
df.select(df["column1"], df["column2"]).show()
> This is useful when you want to select columns with special characters in their names

## range of columns
df.select("column1":"column5").show()
> This is useful when you want to select a range of columns
df.select(df.columns[1:5]).show()
> This is useful when you want to select a range of columns by index

## How to select all columns
df6 = df.select("*")
df6.show()


>> Pyspark withColumn() is a transformation function that is used to change the value, 
convert the datatype of the existing columns, add new columns, and many more uses.

df = spark.createDataFrame([(1, "Alice", 25), (2, "Bob", 30)], ["id", "name", "age"])
df.withColumn("age", df["age"] + 1).show()

> Add a column 
df.withColumn("new_column", df["column1"] + df["column2"]).show()



withColumnRenamed():
use Pyspark withColumnRenamed() to rename an existing column in a DataFrame.

> Create Dataframe
> Add new column based on existing column
> Add new column
> change datatype
> Update the value of an existing column 

> Add a new column based on constant value 

from pyspark.sql.functions import lit
df.withColumn("new_column", lit("constant_value")).show()
# lit is used to add a constant value to a column

> Change the schema of a column
df.withColumn("age", df["age"].cast("string")).show()
# cast is used to change the data type of a column

> From pyspark.sql.functions import col
df.withColumn("age", col("age").cast("string")).show()
# col is used to reference a column

> Rename a column
df.withColumnRenamed("old_name", "new_name").show()
# withColumnRenamed() is used to rename a column


> Filter in pyspark
Pyspark filter() function is used to filter the rows in DataFrame based on certain conditions.
you can also use where() function which is an alias for filter().

df.filter(df["column1"] > 10).show()
# filter() is used to filter rows based on a condition

df.where(df["column1"] > 10).show()
# where() is used to filter rows based on a condition

> df.filter(df.address=="India").show()
> df.where(df.address!="India").show()

> df.filter((df.age > 25) & (df.name == "Alice")).show()
# & is used for AND operation
# | is used for OR operation
# ~ is used for NOT operation

df.filter(df.name.startswith("A")).show()
df.filter(df.name.endswith("e")).show()
df.filter(df.name.contains("li")).show()
df.filter(df.name.isin(["Alice", "Bob"])).show()
df.filter(df.name.isNull()).show()
df.filter(df.name.isNotNull()).show()
df.filter(df.name.like("%li%")).show()
