format ==> Date file format
   CSV,JSON,PARQUET,AVRO,JDBC,TABLE

option => inferSchema => true/false
         => header => true/false
         => mode 

Schema => manual schema =>
           StructType, StructField

DataframeReader API 
       -> HOW TO ACCESS ?
         -> spark.read
         -> spark.read.format("csv")
         -> spark.read.option("header", "true")
         -> spark.read.option("inferSchema", "true")
         -> spark.read.option("mode", "DROPMALFORMED"/"PERMISSIVE"/"FAILFAST")
         -> spark.read.schema("schema")
         -> spark.read.load("path")

FailFast => throws exception if there is a bad record
DropMalformed => drops the bad records
Permissive => sets all the fields to null if there is a bad record
-> Permissive is default mode

from pyspark.sql.types import StructType, StructField, StringType, IntegerType, DoubleType

schema = StructType([
    StructField("ID", IntegerType(), True),
    StructField("name", StringType(), True),
    StructField("age", IntegerType(), True),
    StructField("salary", DoubleType(), True)
])

flight_df_header_schema = spark.read.format("csv") \
    .option("header","false") \
    .option("skipRows",1) \
    .option("inferSchema","false") \
    .schema(schema) \
    .load("file:/Workspace/Users/gulati.r7121@gmail.com/Drafts/myflights.csv")

flight_df_header_schema.printSchema()
