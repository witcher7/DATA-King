## INPUT FUNCTION is used to get input from the user
name = input("Enter your name: ")

# By default, input is always a string
age = input("Enter your age: ")
print("You are " + age + " years old")

# To convert input to integer
age = int(input("Enter your age: "))
print("You are " + str(age) + " years old")

# To convert input to float
height = float(input("Enter your height: "))
print("You are " + str(height) + " cm tall")

## LOOP 
FOR LOOP
WHILE LOOP
DO WHILE LOOP

# for loop 
for i in range(10):
    print(i)

# while loop 
i = 0
while i < 10:
    print(i)
    i += 1

# do while loop 
i = 0
while True:
    print(i)
    i += 1
    if i >= 10:
        break

## Break is used to exit the loop
## Continue is used to skip the current iteration and move to the next iteration
## Pass is used to do nothing

for i in range(10):
    if i == 5:
        continue
    print(i)

for i in range(10):
    if i == 5:
        pass
    print(i)

for i in range(10):
    if i == 5:
        break
    print(i)
