# To print 
print("Hello World")

## DATA TYPES AND VARIABLES
name = "John"
age = 30
height = 5.9
is_student = True

## PRINTING VARIABLES
print(name)
print(age)
print(height)
print(is_student)

## TYPE CONVERSION
age = str(age)
print(type(age))
print("Age is: " + age)
print(f"Age is: {age}")

# STATIC TYPING VS DYNAMIC TYPING
# STATIC TYPING: Type is known at compile time
# DYNAMIC TYPING: Type is known at runtime
for example in python, the type is known at runtime
b = 5
b = "hello"
print(b)
